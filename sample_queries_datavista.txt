-- Hires this month
SELECT TOP (50)
  CANDIDATEFIRSTNAME,
  CANDIDATELASTNAME,
  JOBTITLE,
  COMPANYNAME,
  PRIMARYRECRUITERNAME,
  PLACEMENTDATE,
  STARTDATE,
  Location
FROM DataVista.dbo.CR_HireMaster
WHERE TRY_CONVERT(date, PLACEMENTDATE) >= DATEFROMPARTS(YEAR(GETDATE()), MONTH(GETDATE()), 1)
ORDER BY TRY_CONVERT(date, PLACEMENTDATE) DESC;

-- Candidates recruited / worked by a user (search all stages)
SELECT TOP (100)
  Stage,
  CANDIDATEFIRSTNAME,
  CANDIDATELASTNAME,
  JOBTITLE,
  COMPANYNAME,
  AGREEDPAYRATE,
  Location,
  ActivityDate,
  PRIMARYRECRUITERNAME
FROM (
  SELECT
    'Submittal' AS Stage,
    CANDIDATEFIRSTNAME, CANDIDATELASTNAME, JOBTITLE, COMPANYNAME, AGREEDPAYRATE, Location,
    SUBMITTALDATE AS ActivityDate, PRIMARYRECRUITERNAME, USERFIRSTNAME, USERLASTNAME, userid
  FROM DataVista.dbo.CR_SubmittalMaster
  UNION ALL
  SELECT
    'Interview', CANDIDATEFIRSTNAME, CANDIDATELASTNAME, JOBTITLE, COMPANYNAME, AGREEDPAYRATE, Location,
    INTERVIEWDATE, PRIMARYRECRUITERNAME, USERFIRSTNAME, USERLASTNAME, userid
  FROM DataVista.dbo.CR_InterviewMaster
  UNION ALL
  SELECT
    'Hire', CANDIDATEFIRSTNAME, CANDIDATELASTNAME, JOBTITLE, COMPANYNAME, AGREEDPAYRATE, Location,
    PLACEMENTDATE, PRIMARYRECRUITERNAME, USERFIRSTNAME, USERLASTNAME, userid
  FROM DataVista.dbo.CR_HireMaster
  UNION ALL
  SELECT
    'Reject', CANDIDATEFIRSTNAME, CANDIDATELASTNAME, JOBTITLE, COMPANYNAME, AGREEDPAYRATE, Location,
    COALESCE(INTERNALREJECTDATE, EXTERNALREJECTDATE), PRIMARYRECRUITERNAME, USERFIRSTNAME, USERLASTNAME, userid
  FROM DataVista.dbo.CR_RejectMaster
) AS pipeline
WHERE PRIMARYRECRUITERNAME LIKE '%Jordan%'
   OR (USERFIRSTNAME LIKE '%Jordan%' AND USERLASTNAME LIKE '%Lee%')
   OR (USERFIRSTNAME + ' ' + USERLASTNAME) LIKE '%Jordan%Lee%'
ORDER BY Stage, CANDIDATELASTNAME, CANDIDATEFIRSTNAME;

-- Agreed pay rate + company for a candidate when stage is unknown (do NOT assume Hire)
SELECT Stage, CANDIDATEFIRSTNAME, CANDIDATELASTNAME, COMPANYNAME, AGREEDPAYRATE, JOBTITLE, Location
FROM (
  SELECT 'Submittal' AS Stage, CANDIDATEFIRSTNAME, CANDIDATELASTNAME, COMPANYNAME, AGREEDPAYRATE, JOBTITLE, Location
  FROM DataVista.dbo.CR_SubmittalMaster
  WHERE CANDIDATEFIRSTNAME LIKE '%Akshay%' AND CANDIDATELASTNAME LIKE '%Soni%'
  UNION ALL
  SELECT 'Interview', CANDIDATEFIRSTNAME, CANDIDATELASTNAME, COMPANYNAME, AGREEDPAYRATE, JOBTITLE, Location
  FROM DataVista.dbo.CR_InterviewMaster
  WHERE CANDIDATEFIRSTNAME LIKE '%Akshay%' AND CANDIDATELASTNAME LIKE '%Soni%'
  UNION ALL
  SELECT 'Hire', CANDIDATEFIRSTNAME, CANDIDATELASTNAME, COMPANYNAME, AGREEDPAYRATE, JOBTITLE, Location
  FROM DataVista.dbo.CR_HireMaster
  WHERE CANDIDATEFIRSTNAME LIKE '%Akshay%' AND CANDIDATELASTNAME LIKE '%Soni%'
  UNION ALL
  SELECT 'Reject', CANDIDATEFIRSTNAME, CANDIDATELASTNAME, COMPANYNAME, AGREEDPAYRATE, JOBTITLE, Location
  FROM DataVista.dbo.CR_RejectMaster
  WHERE CANDIDATEFIRSTNAME LIKE '%Akshay%' AND CANDIDATELASTNAME LIKE '%Soni%'
) AS pipeline
WHERE NULLIF(LTRIM(RTRIM(COMPANYNAME)), '') IS NOT NULL
   OR NULLIF(LTRIM(RTRIM(AGREEDPAYRATE)), '') IS NOT NULL
ORDER BY Stage;

-- How many submittals did someone make this month?
SELECT COUNT(*) AS submittal_count
FROM DataVista.dbo.CR_SubmittalMaster
WHERE TRY_CONVERT(date, SUBMITTALDATE) >= DATEFROMPARTS(YEAR(GETDATE()), MONTH(GETDATE()), 1)
  AND (
    PRIMARYRECRUITERNAME LIKE '%Jordan%'
    OR (USERFIRSTNAME LIKE '%Jordan%' AND USERLASTNAME LIKE '%Lee%')
    OR (USERFIRSTNAME + ' ' + USERLASTNAME) LIKE '%Jordan%Lee%'
  );

-- Submittal pay rate + company (when question says submittal)

  COMPANYNAME,
  AGREEDPAYRATE,
  JOBTITLE,
  SUBMITTALDATE,

-- Interviews for a candidate by first name

  CANDIDATEFIRSTNAME,
  CANDIDATELASTNAME,
  JOBTITLE,
  COMPANYNAME,
  INTERVIEWDATE,
  INTERVIEWFLAG,
  PRIMARYRECRUITERNAME
FROM DataVista.dbo.CR_InterviewMaster

-- Reject reasons this year
SELECT
  REJECTREASON,
  COUNT(*) AS reject_count
FROM DataVista.dbo.CR_RejectMaster
WHERE TRY_CONVERT(date, COALESCE(INTERNALREJECTDATE, EXTERNALREJECTDATE)) >= DATEFROMPARTS(YEAR(GETDATE()), 1, 1)
GROUP BY REJECTREASON
ORDER BY reject_count DESC;

-- Placement date (offer received) vs start date (first day of work)
SELECT
  CANDIDATEFIRSTNAME,
  CANDIDATELASTNAME,
  COMPANYNAME,
  JOBTITLE,
  PLACEMENTDATE AS offer_or_placement_date,
  STARTDATE AS work_start_date,
  PRIMARYRECRUITERNAME
FROM DataVista.dbo.CR_HireMaster
WHERE CANDIDATEFIRSTNAME LIKE '%John%'
  AND CANDIDATELASTNAME LIKE '%Smith%';

-- How many clients/companies a recruiter got this month (performance)
SELECT
  COALESCE(PRIMARYRECRUITERNAME, USERFIRSTNAME + ' ' + USERLASTNAME) AS recruiter_name,
  COUNT(DISTINCT COMPANYNAME) AS client_count,
  COUNT(DISTINCT CANDIDATEID) AS candidate_count
FROM DataVista.dbo.CR_HireMaster
WHERE (
        PRIMARYRECRUITERNAME LIKE '%Jordan%'
        OR (USERFIRSTNAME LIKE '%Jordan%' AND USERLASTNAME LIKE '%Lee%')
      )
  AND TRY_CONVERT(date, PLACEMENTDATE) >= DATEFROMPARTS(YEAR(GETDATE()), MONTH(GETDATE()), 1)
GROUP BY COALESCE(PRIMARYRECRUITERNAME, USERFIRSTNAME + ' ' + USERLASTNAME);
